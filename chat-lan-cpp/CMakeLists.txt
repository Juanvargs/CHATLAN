cmake_minimum_required(VERSION 3.22)
project(ChatLAN VERSION 0.2.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
set(CMAKE_AUTORCC ON)
find_package(Qt6 6.4 REQUIRED COMPONENTS Core Network Widgets Test)
find_package(OpenSSL REQUIRED)
find_package(SQLite3 REQUIRED)
add_library(chat_common src/common/protocol.cpp src/common/crypto.cpp src/common/paths.cpp src/common/store.cpp)
target_include_directories(chat_common PUBLIC src)
target_link_libraries(chat_common PUBLIC Qt6::Core Qt6::Network OpenSSL::Crypto SQLite::SQLite3)
target_compile_options(chat_common PRIVATE -Wall -Wextra -Wpedantic)
add_library(chat_server src/server/server.cpp)
target_link_libraries(chat_server PUBLIC chat_common)
add_library(chat_client src/client/connection.cpp)
target_link_libraries(chat_client PUBLIC chat_common)
add_executable(chat-lan-server-cpp src/server/main.cpp)
target_link_libraries(chat-lan-server-cpp PRIVATE chat_server)
add_executable(chat-lan-admin-cpp src/admin/main.cpp)
target_link_libraries(chat-lan-admin-cpp PRIVATE chat_common)
add_executable(ChatLANCpp src/client/main.cpp src/client/window.cpp resources.qrc)
target_link_libraries(ChatLANCpp PRIVATE chat_client Qt6::Widgets)
add_executable(chat-lan-admin-gui-cpp src/admin/gui.cpp resources.qrc)
target_link_libraries(chat-lan-admin-gui-cpp PRIVATE chat_common Qt6::Widgets)
include(CTest)
if(BUILD_TESTING)
  add_executable(chat-tests tests/test_chat.cpp)
  target_link_libraries(chat-tests PRIVATE chat_server chat_client Qt6::Test)
  add_test(NAME chat-core COMMAND chat-tests)
  add_executable(chat-ui-tests tests/test_ui.cpp src/client/window.cpp resources.qrc)
  target_link_libraries(chat-ui-tests PRIVATE chat_client chat_server Qt6::Widgets Qt6::Test)
  add_test(NAME chat-ui COMMAND chat-ui-tests)
  set_tests_properties(chat-ui PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
  add_executable(chat-admin-ui-tests tests/test_admin_ui.cpp resources.qrc)
  target_link_libraries(chat-admin-ui-tests PRIVATE chat_common Qt6::Widgets Qt6::Test)
  add_test(NAME chat-admin-ui COMMAND chat-admin-ui-tests)
  set_tests_properties(chat-admin-ui PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
endif()
install(TARGETS ChatLANCpp chat-lan-server-cpp chat-lan-admin-cpp chat-lan-admin-gui-cpp RUNTIME DESTINATION bin)
install(FILES packaging/chat-lan-cpp.desktop packaging/chat-lan-cpp-admin.desktop DESTINATION share/applications)
install(FILES assets/chat-lan-logo.png DESTINATION share/icons/hicolor/256x256/apps RENAME chat-lan-cpp.png)
install(FILES packaging/chat-lan-server-cpp.service DESTINATION lib/systemd/user)
